'use strict';

const Docker = require('dockerode');
const crypto = require('crypto');

const docker = new Docker();

const TARGET_IMAGE = 'node:20-alpine';
const ATTACK_IMAGE = 'node:20-alpine';

const activeSandboxes = new Map();

function createSandboxId() {
  return crypto.randomBytes(6).toString('hex');
}

function createNames(id) {
  return {
    network: `sentinel-network-${id}`,
    target: `sentinel-target-${id}`,
    attacker: `sentinel-attacker-${id}`,
  };
}

function safeError(error) {
  return error instanceof Error
    ? error.message
    : String(error);
}

async function imageExists(imageName) {
  try {
    await docker.getImage(imageName).inspect();
    return true;
  } catch {
    return false;
  }
}

async function pullImage(imageName) {
  if (await imageExists(imageName)) {
    return;
  }

  console.log(`[sandbox] Pulling ${imageName}...`);

  const stream = await docker.pull(imageName);

  await new Promise((resolve, reject) => {
    docker.modem.followProgress(
      stream,
      error => {
        if (error) {
          reject(error);
          return;
        }

        resolve();
      }
    );
  });
}

async function checkDocker() {
  try {
    const info = await docker.info();

    return {
      ok: true,
      version: info.ServerVersion,
    };
  } catch (error) {
    return {
      ok: false,
      reason:
        `Docker is unavailable: ${safeError(error)}`,
    };
  }
}

/* ------------------------------------------------------------------ */
/*  Classify findings into attack categories                           */
/* ------------------------------------------------------------------ */

function classifyFindings(findings) {
  const categories = {
    sqli: [],
    xss: [],
    cmdi: [],
    path_traversal: [],
    auth_bypass: [],
    code_injection: [],
    other: [],
  };

  for (const finding of findings) {
    // Prefer LLM-assigned attackType, fall back to keyword matching.
    let category = finding.attackType || '';

    if (!categories[category]) {
      // Fall back to keyword-based detection.
      const text = [
        finding.type, finding.explanation, finding.fix,
      ].join(' ').toLowerCase();

      if (/sql|database injection|query injection/.test(text)) {
        category = 'sqli';
      } else if (/xss|cross.?site scripting|html injection/.test(text)) {
        category = 'xss';
      } else if (/command injection|shell injection|exec|child_process/.test(text)) {
        category = 'cmdi';
      } else if (/path traversal|directory traversal/.test(text)) {
        category = 'path_traversal';
      } else if (/authentication bypass|unauthenticated|missing auth/.test(text)) {
        category = 'auth_bypass';
      } else if (/eval|code injection/.test(text)) {
        category = 'code_injection';
      } else {
        category = 'other';
      }
    }

    if (!categories[category]) {
      categories[category] = [];
    }

    categories[category].push(finding);
  }

  return categories;
}

/* ------------------------------------------------------------------ */
/*  Build the target application — uses LLM-generated vulnerable code  */
/* ------------------------------------------------------------------ */

function buildTargetApplication(findings, categories, useRealCode = false) {
  const routes = [];

  // Check if any findings have LLM-generated vulnerable code or real code
  const hasLLMVulnerableCode = findings.some(f => f.vulnerableCode && f.vulnerableCode.trim());
  const hasRealCode = findings.some(f => f.vulnerableCode && f.vulnerableCode.trim() && useRealCode);

  if (hasRealCode) {
    // Use real (sanitized) code from findings
    const findingWithCode = findings.find(f => f.vulnerableCode && f.vulnerableCode.trim());
    routes.push(`
  // Real code endpoint (sanitized)
  if (url.pathname === '/real-vulnerable') {
    try {
      ${findingWithCode.vulnerableCode}
      return json(res, 200, { 
        vulnerable: true,
        finding: ${JSON.stringify(findingWithCode.type || 'Real code vulnerability')},
        mode: 'real-code'
      });
    } catch (error) {
      return json(res, 500, { error: error.message });
    }
  }`);
  } else if (hasLLMVulnerableCode) {
    // Use LLM-generated vulnerable code for a more realistic target
    const findingWithCode = findings.find(f => f.vulnerableCode && f.vulnerableCode.trim());
    routes.push(`
  // LLM-generated vulnerable endpoint
  if (url.pathname === '/llm-vulnerable') {
    try {
      ${findingWithCode.vulnerableCode}
      return json(res, 200, { 
        vulnerable: true,
        finding: ${JSON.stringify(findingWithCode.type || 'LLM-generated vulnerability')}
      });
    } catch (error) {
      return json(res, 500, { error: error.message });
    }
  }`);
  }

  // SQL Injection endpoints
  if (categories.sqli.length > 0) {
    const f = categories.sqli[0];
    routes.push(`
  if (url.pathname === '/search') {
    const query = url.searchParams.get('q') || '';
    // Vulnerable: string concatenation in SQL query
    const simulatedQuery = "SELECT * FROM users WHERE name = '" + query + "'";
    const injectionDetected =
      query.includes("' OR ") || query.includes("'--") ||
      query.includes("'; ") || query.includes("1=1");
    return json(res, 200, {
      query: simulatedQuery,
      vulnerable: injectionDetected,
      results: injectionDetected ? users : [],
      finding: ${JSON.stringify(f.type || 'SQL Injection')}
    });
  }`);
  }

  // XSS endpoints
  if (categories.xss.length > 0) {
    const f = categories.xss[0];
    routes.push(`
  if (url.pathname === '/greet') {
    const name = url.searchParams.get('name') || 'World';
    // Vulnerable: unescaped user input in HTML
    return html(res, 200,
      '<html><body>Hello ' + name + '</body></html>'
    );
  }`);
  }

  // Command Injection endpoints
  if (categories.cmdi.length > 0) {
    routes.push(`
  if (url.pathname === '/ping') {
    const host = url.searchParams.get('host') || 'localhost';
    const suspicious =
      host.includes(';') || host.includes('&&') ||
      host.includes('|') || host.includes('$(');
    return json(res, 200, {
      vulnerable: suspicious,
      simulatedCommand: 'ping -c 1 ' + host,
      note: 'Command is NOT executed. This is a controlled simulation.'
    });
  }`);
  }

  // Path Traversal endpoints
  if (categories.path_traversal.length > 0) {
    routes.push(`
  if (url.pathname === '/file') {
    const requested = url.searchParams.get('name') || 'public.txt';
    return json(res, 200, {
      vulnerable: requested.includes('..'),
      requested,
      content: files[requested] || 'File not found'
    });
  }`);
  }

  // Auth Bypass endpoints
  if (categories.auth_bypass.length > 0) {
    routes.push(`
  if (url.pathname === '/admin') {
    return json(res, 200, {
      vulnerable: true,
      message: 'Simulated protected administrator data',
      users
    });
  }`);
  }

  // Code Injection endpoints
  if (categories.code_injection.length > 0) {
    routes.push(`
  if (url.pathname === '/eval') {
    const code = url.searchParams.get('code') || '';
    return json(res, 200, {
      vulnerable: code.length > 0,
      input: code,
      note: 'eval() is NOT called. This is a controlled simulation.'
    });
  }`);
  }

  return `
const http = require('http');
const { URL } = require('url');

const users = [
  { id: 1, username: 'admin', email: 'admin@example.local', role: 'administrator' },
  { id: 2, username: 'student', email: 'student@example.local', role: 'user' }
];

const files = {
  'public.txt': 'This is a public demonstration file.',
  '../secret.txt': 'SIMULATED_SECRET_VALUE'
};

function json(res, status, body) {
  res.writeHead(status, { 'Content-Type': 'application/json' });
  res.end(JSON.stringify(body));
}

function html(res, status, body) {
  res.writeHead(status, { 'Content-Type': 'text/html; charset=utf-8' });
  res.end(body);
}

const server = http.createServer((req, res) => {
  const url = new URL(req.url, 'http://localhost');

  if (url.pathname === '/health') {
    return json(res, 200, { status: 'ok' });
  }

  ${routes.join('\n')}

  json(res, 404, { error: 'Not found' });
});

server.listen(8080, '0.0.0.0', () => {
  console.log('SentinelAI target listening on port 8080');
});
`;
}

/* ------------------------------------------------------------------ */
/*  Build the attacker script — uses LLM-generated payloads           */
/* ------------------------------------------------------------------ */

function buildAttackerScript(findings, categories) {
  const attacks = [];

  // Check if any findings have LLM-generated vulnerable code
  const hasLLMVulnerableCode = findings.some(f => f.vulnerableCode && f.vulnerableCode.trim());

  // For each finding that has LLM-generated payloads, create targeted attacks.
  for (const finding of findings) {
    const payloads = finding.attackPayloads || [];
    const type = finding.attackType || 'other';
    const label = finding.type || finding.vulnerability || 'Unknown';

    if (payloads.length > 0) {
      // Use LLM-generated payloads — these are targeted to the user's code.
      for (const payload of payloads) {
        let path = '';
        const encoded = encodeURIComponent(payload);

        // If LLM vulnerable code exists, target that endpoint
        if (hasLLMVulnerableCode) {
          path = '/llm-vulnerable?payload=' + encoded;
        } else {
          // Map attack type to the target endpoint.
          switch (type) {
            case 'sqli':
              path = '/search?q=' + encoded;
              break;
            case 'xss':
              path = '/greet?name=' + encoded;
              break;
            case 'cmdi':
              path = '/ping?host=' + encoded;
              break;
            case 'path_traversal':
              path = '/file?name=' + encoded;
              break;
            case 'auth_bypass':
              path = '/admin';
              break;
            case 'code_injection':
              path = '/eval?code=' + encoded;
              break;
            default:
              path = '/search?q=' + encoded;
          }
        }

        const isXss = type === 'xss';

        attacks.push(`
await test(
  ${JSON.stringify(label + ' [LLM payload]')},
  '${path.replace(/'/g, "\\'")}',
  body => ${isXss
            ? `String(body).includes(${JSON.stringify(payload)})`
            : 'body.vulnerable === true'},
  ${JSON.stringify(payload)}${isXss ? ',\n  true' : ''}
);`);
      }
    }
  }

  // Fallback: if no LLM payloads, use generic attacks based on categories.
  if (attacks.length === 0) {
    if (categories.sqli.length > 0) {
      attacks.push(`
await test(
  'SQL Injection [generic]',
  '/search?q=' + encodeURIComponent("' OR '1'='1"),
  body => body.vulnerable === true,
  "' OR '1'='1"
);`);
    }

    if (categories.xss.length > 0) {
      attacks.push(`
await test(
  'Reflected XSS [generic]',
  '/greet?name=' + encodeURIComponent('<script>alert(1)</script>'),
  body => String(body).includes('<script>alert(1)</script>'),
  '<script>alert(1)</script>',
  true
);`);
    }

    if (categories.cmdi.length > 0) {
      attacks.push(`
await test(
  'Command Injection [generic]',
  '/ping?host=' + encodeURIComponent('localhost;whoami'),
  body => body.vulnerable === true,
  'localhost;whoami'
);`);
    }

    if (categories.path_traversal.length > 0) {
      attacks.push(`
await test(
  'Path Traversal [generic]',
  '/file?name=' + encodeURIComponent('../secret.txt'),
  body => body.vulnerable === true,
  '../secret.txt'
);`);
    }

    if (categories.auth_bypass.length > 0) {
      attacks.push(`
await test(
  'Authentication Bypass [generic]',
  '/admin',
  body => body.vulnerable === true,
  'Unauthenticated GET /admin'
);`);
    }
  }

  if (attacks.length === 0) {
    attacks.push(`
results.push({
  tool: 'SentinelAI Simulator',
  target: 'Generated local target',
  payload: '',
  output: 'No supported attack category matched the findings.',
  evidence: 'Static findings were preserved, but no controlled simulation was available.',
  success: false
});`);
  }

  return `
const http = require('http');

const results = [];

function request(path) {
  return new Promise((resolve, reject) => {
    const req = http.get(
      {
        hostname: 'target',
        port: 8080,
        path,
        timeout: 10000
      },
      res => {
        let data = '';
        res.on('data', chunk => { data += chunk.toString(); });
        res.on('end', () => {
          resolve({ statusCode: res.statusCode, data });
        });
      }
    );

    req.on('error', reject);
    req.on('timeout', () => {
      req.destroy(new Error('Request timed out'));
    });
  });
}

async function test(tool, path, predicate, payload, raw = false) {
  try {
    const response = await request(path);

    let parsed = response.data;

    if (!raw) {
      try {
        parsed = JSON.parse(response.data);
      } catch {
        parsed = response.data;
      }
    }

    const success = Boolean(predicate(parsed));

    results.push({
      tool,
      target: 'http://target:8080' + path,
      payload,
      output: response.data,
      evidence: success
        ? 'The controlled target reflected the vulnerable behaviour.'
        : 'The controlled target did not exhibit the vulnerable behaviour.',
      success
    });
  } catch (error) {
    results.push({
      tool,
      target: 'http://target:8080' + path,
      payload,
      output: '',
      evidence: 'Attack simulation failed: ' + error.message,
      success: false
    });
  }
}

async function main() {
  ${attacks.join('\n')}

  process.stdout.write(JSON.stringify(results));
}

main().catch(error => {
  console.error(error);
  process.exit(1);
});
`;
}

/* ------------------------------------------------------------------ */
/*  Container helpers                                                  */
/* ------------------------------------------------------------------ */

async function waitForContainer(container) {
  const result = await container.wait();
  return result.StatusCode;
}

async function getContainerLogs(container) {
  const buffer = await container.logs({
    stdout: true,
    stderr: true,
  });

  return buffer
    .toString('utf8')
    .replace(
      /[\u0000-\u0008\u000B\u000C\u000E-\u001F]/g,
      ''
    );
}

async function waitForTarget() {
  await new Promise(resolve => {
    setTimeout(resolve, 2500);
  });
}

async function removeContainer(container) {
  if (!container) return;

  try {
    await container.remove({ force: true });
  } catch {
    // Container may already have been removed.
  }
}

async function removeNetwork(network) {
  if (!network) return;

  try {
    await network.remove();
  } catch {
    // Network may already have been removed.
  }
}

async function cleanupSandbox(state) {
  if (!state) return;

  await Promise.allSettled([
    removeContainer(state.attackerContainer),
    removeContainer(state.targetContainer),
  ]);

  await removeNetwork(state.network);
}

function parseAttackOutput(output) {
  const start = output.indexOf('[');
  const end = output.lastIndexOf(']');

  if (start === -1 || end === -1) {
    throw new Error(
      `Could not parse sandbox output: ${output}`
    );
  }

  return JSON.parse(
    output.slice(start, end + 1)
  );
}

/* ------------------------------------------------------------------ */
/*  Main sandbox runner                                                */
/* ------------------------------------------------------------------ */

async function runSandbox(findings, options = {}) {
  if (!Array.isArray(findings)) {
    throw new Error('Findings must be an array.');
  }

  const sandboxId = createSandboxId();
  const names = createNames(sandboxId);

  const startedAt = new Date().toISOString();
  const events = [];

  // Options: mode ('simulation' | 'real'), useRealCode
  const mode = options.mode || 'simulation';
  const useRealCode = options.useRealCode || mode === 'real';

  // Step 1: Sandbox initialization
  events.push({
    step: 1,
    description: `Sandbox started - Mode: ${useRealCode ? 'Real Code Attack' : 'Safe Simulation'}`,
    timestamp: new Date().toISOString()
  });

  const categories = classifyFindings(findings);
  const hasLLMPayloads = findings.some(
    f => Array.isArray(f.attackPayloads) && f.attackPayloads.length > 0
  );

  console.log(
    `[sandbox] ${findings.length} findings, ` +
    `Mode: ${mode}, ` +
    `LLM payloads: ${hasLLMPayloads ? 'yes' : 'no (using generic)'}`
  );

  const state = {
    sandboxId,
    network: null,
    targetContainer: null,
    attackerContainer: null,
  };

  activeSandboxes.set(sandboxId, state);

  try {
    await Promise.all([
      pullImage(TARGET_IMAGE),
      pullImage(ATTACK_IMAGE),
    ]);

    state.network = await docker.createNetwork({
      Name: names.network,
      Driver: 'bridge',
      Internal: true,
    });

    // Step 2: Network created
    events.push({
      step: 2,
      description: 'Isolated network created for sandbox containers',
      timestamp: new Date().toISOString()
    });

    const targetCode = buildTargetApplication(findings, categories, useRealCode);

    state.targetContainer = await docker.createContainer({
      name: names.target,
      Image: TARGET_IMAGE,
      Env: [
        `TARGET_CODE=${Buffer.from(targetCode, 'utf8').toString('base64')}`,
      ],
      Cmd: [
        'sh', '-c',
        'echo "$TARGET_CODE" | base64 -d > /tmp/target.js && node /tmp/target.js',
      ],
      HostConfig: {
        NetworkMode: names.network,
        Memory: 256 * 1024 * 1024,
        NanoCpus: 500_000_000,
        PidsLimit: 64,
        CapDrop: ['ALL'],
        SecurityOpt: ['no-new-privileges:true'],
        ReadonlyRootfs: true,
        Tmpfs: { '/tmp': 'rw,noexec,nosuid,size=16m' },
      },
      NetworkingConfig: {
        EndpointsConfig: {
          [names.network]: { Aliases: ['target'] },
        },
      },
    });

    await state.targetContainer.start();
    await waitForTarget();

    // Step 3: Target container started with vulnerable application
    events.push({
      step: 3,
      description: 'Target container started - Vulnerable application deployed',
      mockFrontend: {
        html: generateMockFrontendHTML(findings, categories),
        vulnerabilities: findings.map(f => ({
          type: f.type || 'Unknown',
          severity: f.severity || 'Medium',
          file: f.file || 'Unknown文件'
        }))
      },
      timestamp: new Date().toISOString()
    });

    const attackerCode = buildAttackerScript(findings, categories);

    state.attackerContainer = await docker.createContainer({
      name: names.attacker,
      Image: ATTACK_IMAGE,
      Env: [
        `ATTACK_CODE=${Buffer.from(attackerCode, 'utf8').toString('base64')}`,
      ],
      Cmd: [
        'sh', '-c',
        'echo "$ATTACK_CODE" | base64 -d > /tmp/attacker.js && node /tmp/attacker.js',
      ],
      HostConfig: {
        NetworkMode: names.network,
        Memory: 256 * 1024 * 1024,
        NanoCpus: 500_000_000,
        PidsLimit: 64,
        CapDrop: ['ALL'],
        SecurityOpt: ['no-new-privileges:true'],
        ReadonlyRootfs: true,
        Tmpfs: { '/tmp': 'rw,noexec,nosuid,size=16m' },
      },
      NetworkingConfig: {
        EndpointsConfig: {
          [names.network]: {},
        },
      },
    });

    await state.attackerContainer.start();

    // Step 4: Attack execution started
    events.push({
      step: 4,
      description: 'Attack container started - Executing exploit payloads',
      timestamp: new Date().toISOString()
    });

    const statusCode = await waitForContainer(state.attackerContainer);

    const output = await getContainerLogs(state.attackerContainer);

    if (statusCode !== 0) {
      throw new Error(
        `Attacker container exited with code ${statusCode}: ${output}`
      );
    }

    const attacks = parseAttackOutput(output);

    const successfulAttacks = attacks.filter(a => a.success).length;

    // Step 5: Attack results collected
    events.push({
      step: 5,
      description: `Attack execution completed - ${successfulAttacks} of ${attacks.length} attacks successful`,
      dataLeak: attacks.filter(a => a.success).map(a => ({
        type: a.tool,
        payload: a.payload,
        evidence: a.evidence,
        output: a.output
      })),
      timestamp: new Date().toISOString()
    });

    const finishedAt = new Date().toISOString();

    return {
      sandboxId,
      startedAt,
      finishedAt,
      target: 'Generated isolated local demonstration application',
      attacks,
      events,
      summary:
        successfulAttacks > 0
          ? `${successfulAttacks} of ${attacks.length} attack simulations (${hasLLMPayloads ? 'LLM-targeted' : 'generic'}) reproduced vulnerable behaviour.`
          : 'No attack simulation reproduced vulnerable behaviour.',
    };
  } finally {
    await cleanupSandbox(state);
    activeSandboxes.delete(sandboxId);
  }
}

async function stopSandbox() {
  const states = Array.from(activeSandboxes.values());

  await Promise.allSettled(
    states.map(cleanupSandbox)
  );

  activeSandboxes.clear();

  return { stopped: states.length };
}

module.exports = {
  checkDocker,
  runSandbox,
  stopSandbox,
};
